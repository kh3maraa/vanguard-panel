Signature HMAC

Chaque requête doit inclure ces headers :

  • X-App-Name — nom de votre application
  • X-Timestamp — timestamp Unix (secondes)
  • X-Nonce — nonce unique (UUID v4)
  • X-Signature — HMAC-SHA256 hex de ts.nonce.payload

Client SDK (JavaScript)

async function signedFetch(url, appName, appSecret, body) {
  const ts = Math.floor(Date.now() / 1000).toString();
  const nonce = crypto.randomUUID();
  const payload = JSON.stringify(body);
  const enc = new TextEncoder();
  const key = await crypto.subtle.importKey(
    "raw", enc.encode(appSecret),
    { name:"HMAC", hash:"SHA-256" }, false, ["sign"]
  );
  const sig = await crypto.subtle.sign("HMAC", key, enc.encode(`${ts}.${nonce}.${payload}`));
  const hex = [...new Uint8Array(sig)].map(b => b.toString(16).padStart(2,"0")).join("");
  return fetch(url, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "X-App-Name": appName,
      "X-Timestamp": ts,
      "X-Nonce": nonce,
      "X-Signature": hex
    },
    body: payload
  }).then(r => r.json());
}

// Usage :
const r = await signedFetch(
  "{{ request.url_root }}api/v1/license",
  "mon-app", "TON_API_SECRET",
  { key: "VNGD-XXXX-XXXX-XXXX-XXXX-XXXX", hwid: "abc123def456" }
);
console.log(r);

Exemple Python

import time, uuid, hmac, hashlib, json, requests

def signed_call(app_name, app_secret, endpoint, body):
    ts = str(int(time.time()))
    nonce = str(uuid.uuid4())
    payload = json.dumps(body, separators=(",", ":"))
    sig = hmac.new(
        app_secret.encode(),
        f"{ts}.{nonce}.{payload}".encode(),
        hashlib.sha256
    ).hexdigest()
    return requests.post(
        "{{ request.url_root }}api/v1/" + endpoint,
        data=payload,
        headers={
            "Content-Type": "application/json",
            "X-App-Name": app_name,
            "X-Timestamp": ts,
            "X-Nonce": nonce,
            "X-Signature": sig
        }
    ).json()

r = signed_call("mon-app", "TON_API_SECRET", "license", {
    "key": "VNGD-XXXX-XXXX-XXXX-XXXX-XXXX",
    "hwid": "abc123"
})
print(r)
